## Technical Approach

### 1. Create Registry Schema Definition

Create `pennyfarthing-dist/skills/skill-registry.schema.json`:
```json
{
  "$schema": "http://json-schema.org/draft-07/schema#",
  "type": "object",
  "properties": {
    "version": { "type": "string", "pattern": "^\\d+\\.\\d+\\.\\d+$" },
    "skills": {
      "type": "object",
      "additionalProperties": {
        "$ref": "#/definitions/skill"
      }
    }
  },
  "required": ["version", "skills"],
  "definitions": {
    "skill": {
      "type": "object",
      "properties": {
        "name": { "type": "string" },
        "description": { "type": "string" },
        "category": { "enum": ["development", "project-management", "theming", "tools", "ai-llm", "documentation", "benchmarking"] },
        "tags": { "type": "array", "items": { "type": "string" } },
        "version": { "type": "string" },
        "prerequisites": { "type": "array", "items": { "type": "string" } },
        "examples": {
          "type": "array",
          "items": {
            "type": "object",
            "properties": {
              "context": { "type": "string" },
              "invocation": { "type": "string" }
            },
            "required": ["context", "invocation"]
          }
        },
        "anti_patterns": { "type": "array", "items": { "type": "string" } },
        "related_skills": { "type": "array", "items": { "type": "string" } },
        "keywords": { "type": "array", "items": { "type": "string" } }
      },
      "required": ["name", "description", "category", "tags"]
    }
  }
}
```

### 2. Create Registry File

Create `pennyfarthing-dist/skills/skill-registry.yaml`:
```yaml
version: 1.0.0
skills:
  testing:
    name: testing
    description: TDD patterns and test framework guidance
    category: development
    tags: [tdd, testing, quality]
    version: 1.0.0
    prerequisites: []
    examples:
      - context: Running project tests
        invocation: /testing
      - context: Debugging test failures
        invocation: /testing troubleshooting
    anti_patterns:
      - Don't run tests directly - use testing-runner subagent
    related_skills: [dev-patterns, code-review]
    keywords: [test, spec, jest, vitest, pytest, unit, integration]
  # ... all 18 skills cataloged
```
