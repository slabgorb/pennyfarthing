## Technical Approach

### Extract to: scripts/utils/repo-scan.sh (~40 lines)

Create a new utility that provides:

```bash
#!/usr/bin/env bash
# Repo scanning utilities for workflow status checks

source "${PROJECT_ROOT}/scripts/repo-utils.sh"

# scan_repo_git_status REPO_NAME
# Returns: branch, uncommitted count, ahead count
scan_repo_git_status() {
    local repo="$1"
    local repo_path=$(get_repo_path "$repo")
    cd "${PROJECT_ROOT}/${repo_path}"

    local branch=$(git branch --show-current)
    local uncommitted=$(git status --short | wc -l | tr -d ' ')
    local ahead=$(git log origin/develop..HEAD --oneline 2>/dev/null | wc -l | tr -d ' ')

    echo "${repo}|${branch}|${uncommitted}|${ahead}"
}

# scan_all_repos_status
# Returns: pipe-delimited status for all repos
scan_all_repos_status() {
    for repo in $(get_repos); do
        scan_repo_git_status "$repo"
    done
}

# check_repo_pr REPO_NAME BRANCH
# Returns: PR state (merged|open|none)
check_repo_pr() {
    local repo="$1"
    local branch="$2"
    local repo_path=$(get_repo_path "$repo")
    cd "${PROJECT_ROOT}/${repo_path}"
```
