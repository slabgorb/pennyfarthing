2. `.github/workflows/*.yml` → `act` (if installed)
3. `.gitlab-ci.yml` → `gitlab-runner exec`
4. Fallback: `npm test && npm run lint && npm run build`

### Detection Logic Pattern (from check.sh)
```bash
# Justfile recipe detection
has_just_recipe() {
    if [[ -f "justfile" ]] && command -v just &>/dev/null; then
        just --list 2>/dev/null | grep -q "^$recipe " && return 0
    fi
    return 1
}

# Project root detection
find_project_root() {
    local dir="$PWD"
    while [[ ! -d "$dir/.claude" ]] && [[ "$dir" != "/" ]]; do
        dir="$(dirname "$dir")"
    done
    echo "$dir"
}
```
