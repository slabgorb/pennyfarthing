### Discovery Logic

```typescript
function findCyclist(): string {
  // 1. Environment override
  if (process.env.CYCLIST_PATH) return process.env.CYCLIST_PATH;

  // 2. Sibling directory (development)
  const sibling = join(dirname(process.cwd()), 'cyclist');
  if (existsSync(join(sibling, 'package.json'))) return sibling;

  // 3. Relative to pennyfarthing install
  const relative = join(__dirname, '../../cyclist');
  if (existsSync(join(relative, 'package.json'))) return relative;

  throw new Error('Cyclist not found. Set CYCLIST_PATH or install at ../cyclist');
}
```
