### 1. Create scripts/utils/retry.sh (~60 lines)
```bash
#!/usr/bin/env bash
# Retry with exponential backoff

retry_with_backoff() {
    local max_attempts=${1:-3}
    local delay=${2:-1}
    local max_delay=${3:-30}
    shift 3

    local attempt=1
    while true; do
        if "$@"; then
            return 0
        fi

        if ((attempt >= max_attempts)); then
            return 1
        fi

        sleep "$delay"
        delay=$((delay * 2))
        ((delay > max_delay)) && delay=$max_delay
        ((attempt++))
    done
}

# command_with_fallback PRIMARY_CMD FALLBACK_CMD
command_with_fallback() {
    if eval "$1"; then
        return 0
    else
        eval "$2"
    fi
}
```

### 2. Create scripts/utils/checkpoint.sh (~80 lines)
```bash
#!/usr/bin/env bash
# Session checkpointing

CHECKPOINT_FILE="${PROJECT_ROOT:-.}/.session/checkpoints.log"

checkpoint_save() {
    local label="$1"
    local data="$2"
    local timestamp=$(date -u +"%Y-%m-%dT%H:%M:%SZ")

    mkdir -p "$(dirname "$CHECKPOINT_FILE")"
    echo "${timestamp}|${label}|${data}" >> "$CHECKPOINT_FILE"
}

checkpoint_restore() {
    local label="$1"
    grep "|${label}|" "$CHECKPOINT_FILE" | tail -1 | cut -d'|' -f3
}

checkpoint_list() {
    cat "$CHECKPOINT_FILE" 2>/dev/null | tail -20
}

checkpoint_clear() {
    rm -f "$CHECKPOINT_FILE"
}
```
